function assembleHtmlParts(parts, paths) {
    const total = parts.length;
    return parts.map((part, i) => {
        const dot = spriteConfig.show_nav_accessories ? `<span class="page-dot" style="color:${seededColorFromPath(paths[i])}">●</span>` : '';
        const pageLabel = total > 1 ? `<div class="page-label">${i+1}/${total}${dot}</div>` : '';
        return `<div class="text-page">${part}${pageLabel}</div>`;
    }).join('');
}
